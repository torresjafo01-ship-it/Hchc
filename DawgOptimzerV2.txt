local flagSeen = {}
local flagSet, flagDead = 0, 0

local function applyFlag(name, value)
	if not setF then
		return false
	end
	if getF then
		local ok, cur = pcall(getF, name)
		if not ok or cur == nil then
			if not flagSeen[name] then
				flagSeen[name] = true
				flagDead += 1
			end
			return false
		end
	end
	if pcall(setF, name, tostring(value)) then
		if not flagSeen[name] then
			flagSeen[name] = true
			flagSet += 1
		end
		return true
	end
	return false
end

_G.JAF_LoadOptCopyStartupFlags = function()
	local parts = {}
	for _, f in ipairs(STARTUP_ONLY) do
		parts[#parts + 1] = string.format('  "%s": "%s"', f[1], f[2])
	end
	local json = "{\n" .. table.concat(parts, ",\n") .. "\n}"
	if type(setclipboard) == "function" then
		pcall(setclipboard, json)
		print("[DawgOpt] startup-only flags copied to clipboard -- paste into Bloxstrap > Engine Settings > FastFlags, then relaunch Roblox.")
	else
		print(json)
	end
	return json
end

local KILL_NAMES = {

	BackgroundScreen = true,
	DevGUIBlackoutCurtain = true,
}

local function killIfMatch(inst)
	if inst.Parent == CoreGui and KILL_NAMES[inst.Name] then
		inst:Destroy()
	end
end

local watchConn

if OPT_ON then
	for _, child in ipairs(CoreGui:GetChildren()) do
		killIfMatch(child)
	end
	watchConn = CoreGui.ChildAdded:Connect(killIfMatch)
	pcall(function()
		ReplicatedFirst:RemoveDefaultLoadingScreen()
	end)
end

local loadPhase = true

local function permaStrip(inst)
	if
		inst:IsA("ParticleEmitter")
		or inst:IsA("Trail")
		or inst:IsA("Smoke")
		or inst:IsA("Fire")
		or inst:IsA("Sparkles")
	then
		inst.Enabled = false
	elseif inst:IsA("PostEffect") then
		inst.Enabled = false
	end
end

local function harshStrip(inst)
	if not loadPhase then
		return
	end
	if inst:IsA("Decal") or inst:IsA("Texture") then
		if inst.Transparency ~= 1 then
			inst.Transparency = 1
		end
	elseif inst:IsA("BillboardGui") or inst:IsA("SurfaceGui") then
		if inst.Enabled then
			inst.Enabled = false
		end
	end
end

local function applyPermanentSettings()
	pcall(function()
		settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
	end)
	pcall(function()
		UserSettings():GetService("UserGameSettings").SavedQualityLevel =
			Enum.SavedQualitySetting.QualityLevel1
	end)
	pcall(function()
		Lighting.GlobalShadows = false
		Lighting.EnvironmentDiffuseScale = 0
		Lighting.EnvironmentSpecularScale = 0
	end)
	pcall(function()
		local atmosphere = Lighting:FindFirstChildOfClass("Atmosphere")
		if atmosphere then
			atmosphere.Density = 0
			atmosphere.Glare = 0
			atmosphere.Haze = 0
		end
	end)
	pcall(function()
		local terrain = Workspace:FindFirstChildOfClass("Terrain")
		if terrain then
			terrain.Decoration = false
			terrain.WaterWaveSize = 0
			terrain.WaterWaveSpeed = 0
			terrain.WaterReflectance = 0
			local clouds = terrain:FindFirstChildOfClass("Clouds")
			if clouds then
				clouds.Enabled = false
			end
		end
	end)
end

local coreGuiHidden = false

local function setCoreGui(on)
	pcall(function()
		StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.All, on)
	end)
end

local permaConn, lightingConn

if OPT_ON then
	applyPermanentSettings()

	if COREGUI_ON then
		coreGuiHidden = true
		setCoreGui(false)
	end

	task.spawn(function()
		for _, root in ipairs({ Workspace, Lighting }) do
			local ok, list = pcall(function()
				return root:GetDescendants()
			end)
			if ok then
				local i, n = 1, #list
				while i <= n do
					local t0 = os.clock()
					while i <= n and (os.clock() - t0) < 0.002 do
						pcall(permaStrip, list[i])
						if HARSH_ON then
							pcall(harshStrip, list[i])
						end
						i += 1
					end
					RunService.Heartbeat:Wait()
				end
			end
		end
	end)

	permaConn = Workspace.DescendantAdded:Connect(function(inst)
		pcall(permaStrip, inst)
		if HARSH_ON and loadPhase then
			pcall(harshStrip, inst)
		end
	end)
	lightingConn = Lighting.DescendantAdded:Connect(function(inst)
		pcall(permaStrip, inst)
	end)
end

local finished = false

local function finish(seconds)
	if finished then
		return
	end
	finished = true

	loadPhase = false

	if watchConn and watchConn.Connected then
		watchConn:Disconnect()
	end

	if coreGuiHidden and not KEEP_COREGUI_HIDDEN then
		coreGuiHidden = false
		setCoreGui(true)
	end

	pcall(function()
		meter.finish(seconds)
	end)

	if seconds then
		print(string.format("[DawgOpt] loaded in %.2fs | flags %d set, %d absent | nothing reverted%s", seconds, flagSet, flagDead, BASELINE and " | BASELINE" or ""))
	else
		print("[DawgOpt] load time not measurable (already in-game when the script started)")
	end
end

_G.JAF_LoadOptStop = function()
	if permaConn then
		permaConn:Disconnect()
	end
	if lightingConn then
		lightingConn:Disconnect()
	end
	finish(nil)
end

if LATE then
	task.delay(1, function()
		finish(nil)
	end)
else
	task.spawn(function()
		if not game:IsLoaded() then
			game.Loaded:Wait()
		end
		local lp = Players.LocalPlayer
		if not lp then
			Players:GetPropertyChangedSignal("LocalPlayer"):Wait()
			lp = Players.LocalPlayer
		end
		if lp and not lp.Character then
			lp.CharacterAdded:Wait()
		end
		task.wait(0.1)
		finish(os.clock() - START)
	end)

	task.delay(60, function()
		finish(nil)
	end)
end
